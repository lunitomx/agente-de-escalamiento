### T5 — Integration verification: full People graph traversal
**Risk: LOW** (validation only, no new content)

Verify relationship integrity: every `target` referenced in any People node actually exists as a node ID.

```bash
python3 -c "
import yaml, pathlib

# Collect all node IDs
all_ids = set()
for f in pathlib.Path('.scaleup/knowledge').rglob('*.yaml'):
    try:
        d = yaml.safe_load(f.read_text())
        if 'id' in d:
            all_ids.add(d['id'])
    except: pass

# Check all People relationship targets exist
errors = []
for f in pathlib.Path('.scaleup/knowledge/people').rglob('*.yaml'):
    d = yaml.safe_load(f.read_text())
    for r in d.get('relationships', []):
        if r['target'] not in all_ids:
            errors.append(f'{f.name}: broken target → {r[\"target\"]}')

# Also check decision-people node
dp = yaml.safe_load(open('.scaleup/knowledge/decisions/people.yaml').read())
for r in dp.get('relationships', []):
    if r['target'] not in all_ids:
        errors.append(f'decision-people: broken target → {r[\"target\"]}')

if errors:
    [print(e) for e in errors]
    raise SystemExit(1)

people_nodes = len(list(pathlib.Path('.scaleup/knowledge/people').rglob('*.yaml')))
print(f'Graph integrity OK — {people_nodes} People nodes, 0 broken relationships')
"
```

Size: XS | Dependencies: T1, T2, T3, T4
